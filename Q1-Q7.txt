import json
import re

def analyze_damage_log(lines):
    total = 0
    by_armor = {"front": 0, "left": 0, "right": 0}
    seen_ids = set()
    event_count = 0

    for line in lines:
        if not isinstance(line, str):
            continue
        s = line.strip()
        if not s or s.startswith("#"):
            continue

        # JSON 行
        if s.startswith("{"):
            try:
                data = json.loads(s)
            except Exception:
                continue
            if not isinstance(data, dict):
                continue
            if "armor" not in data or "damage" not in data:
                continue
            armor = data["armor"]
            damage = data["damage"]
            if armor not in ("front", "left", "right"):
                continue
            if not isinstance(damage, int) or isinstance(damage, bool) or damage <= 0:
                continue
            id_val = data.get("id", None)
            if id_val is not None:
                if id_val in seen_ids:
                    continue
                seen_ids.add(id_val)
            total += damage
            by_armor[armor] += damage
            event_count += 1
            continue

        # 传感器行
        segments = s.split(",")
        valid = True
        parsed = []
        for seg in segments:
            seg = seg.strip()
            m = re.fullmatch(r"([FLR]):(\d+)", seg)
            if not m:
                valid = False
                break
            letter, num = m.group(1), int(m.group(2))
            if num <= 0:
                valid = False
                break
            armor = {"F": "front", "L": "left", "R": "right"}[letter]
            parsed.append((armor, num))
        if not valid:
            continue
        for armor, dmg in parsed:
            total += dmg
            by_armor[armor] += dmg
            event_count += 1

    most_hit = None
    if event_count > 0:
        max_dmg = max(by_armor.values())
        if max_dmg > 0:
            for part in ("front", "left", "right"):
                if by_armor[part] == max_dmg:
                    most_hit = part
                    break

    avg = round(total / event_count, 2) if event_count > 0 else 0.0
    return {
        "total": total,
        "by_armor": by_armor,
        "most_hit": most_hit,
        "avg": avg
    }

# ================= Q3: SentryGrid 实现 =================

class SentryGrid:
    def __init__(self, width, height, obstacles, enemy_pos,
                 start_pos=(0, 0), facing=Facing.UP, fuel=100):
        self.width = width
        self.height = height
        self.obstacles = set(obstacles) # 转为 set 提高查找效率
        self.enemy_pos = enemy_pos
        self._current_pos = None
        
        # 规范1：通过调用 setter 来复用校验逻辑
        self.current_pos = start_pos  
        
        self.facing = facing
        self.fuel = fuel
        self.collision_count = 0

    @property
    def current_pos(self):
        return self._current_pos

    @current_pos.setter
    def current_pos(self, value):
        # 规范1：只接受长度为2的 tuple 或 list，其他类型抛出 TypeError
        if not isinstance(value, (tuple, list)) or len(value) != 2:
            raise TypeError("current_pos must be a tuple or list of length 2")
        try:
            # 规范1：元素规范化后以 tuple 存储
            x = int(value[0])
            y = int(value[1])
        except (ValueError, TypeError):
            raise TypeError("current_pos elements must be integers")
        self._current_pos = (x, y)

    def _in_bounds(self, pos):
        """内部辅助函数：检查坐标是否在地图边界内"""
        x, y = pos
        return 0 <= x < self.width and 0 <= y < self.height

    def move_forward(self):
        # 规范4：电量 <= 0 时底盘断电，前进尝试不再产生位移
        if self.fuel <= 0:
            return self.current_pos

        # 规范3：前进消耗 1 单位电量
        self.fuel -= 1

        # 计算前方格子
        dx, dy = self.facing.value
        next_x = self.current_pos[0] + dx
        next_y = self.current_pos[1] + dy
        next_pos = (next_x, next_y)

        # 规范2：前方格为障碍，判定为碰撞——位置不变、朝向不变、collision_count 加 1
        # 注意：这里额外加入了边界检查，避免越界导致死循环或异常
        if next_pos in self.obstacles or not self._in_bounds(next_pos):
            self.collision_count += 1
            return self.current_pos

        # 前方可行则移动到该格
        self._current_pos = next_pos
        # 规范2：返回执行后的当前位
        return self.current_pos

    def turn_left(self):
        # 逆时针旋转：上 -> 左 -> 下 -> 右 -> 上
        order = [Facing.UP, Facing.LEFT, Facing.DOWN, Facing.RIGHT]
        idx = order.index(self.facing)
        self.facing = order[(idx + 1) % 4]
        # 规范5：原地旋转 90°，返回新的 Facing
        return self.facing

    def turn_right(self):
        # 顺时针旋转：上 -> 右 -> 下 -> 左 -> 上
        order = [Facing.UP, Facing.RIGHT, Facing.DOWN, Facing.LEFT]
        idx = order.index(self.facing)
        self.facing = order[(idx + 1) % 4]
        # 规范5：原地旋转 90°，返回新的 Facing
        return self.facing






def next_step_toward(pos, target, obstacles, current_facing=Facing.UP):
    """
    Q4: 贪心导航
    返回下一步应朝向的 Facing。
    """
    x, y = pos
    tx, ty = target
    d0 = abs(tx - x) + abs(ty - y)

    candidates = []
    # 候选方向 = 四邻域中无障碍且移动后曼哈顿距离严格减小的方向
    for f in (Facing.UP, Facing.RIGHT, Facing.DOWN, Facing.LEFT):
        dx, dy = f.value
        nx, ny = x + dx, y + dy
        next_pos = (nx, ny)  
        # 规范1: 相邻格不是障碍
        if next_pos in obstacles:
            continue
            
        # 规范1: 移动到该格后，到目标曼哈顿距离严格减小
        d1 = abs(tx - nx) + abs(ty - ny)
        if d1 < d0:
            candidates.append((f, next_pos))

    # 规范3: 不存在任何候选时，返回 current_facing
    if not candidates:
        return current_facing
        
    # 只有一个候选时直接返回
    if len(candidates) == 1:
        return candidates[0][0]

    # 规范2: 存在多个候选时，先走与目标绝对坐标差较大的轴
    x_diff = abs(tx - x)
    y_diff = abs(ty - y)
    prefer_horizontal = x_diff >= y_diff  # 如果 x 轴距离更大（或相等），优先水平移动

    for f, _ in candidates:
        is_horizontal = f in (Facing.LEFT, Facing.RIGHT)
        if is_horizontal == prefer_horizontal:
            return f

    # 保底 return（理论上不会走到这里）
    return candidates[0][0]

def decide(sensor, state, hp, heat):
    """
    Q5: 哨兵决策机
    读取感知数据、当前状态、血量与热量，返回 (action, new_state)。
    """
    # 1. 非法输入处理：契约外的输入，一律 raise ValueError，不得静默返回任何结果
    required = {"enemy_frames", "enemy_dist", "robot_type", "max_hp"}
    if not isinstance(sensor, dict) or not required.issubset(sensor.keys()):
        raise ValueError("sensor missing required fields")
    if not isinstance(state, SentryState):
        raise ValueError("invalid state")

    frames_raw = sensor["enemy_frames"]
    if not isinstance(frames_raw, (list, tuple)) or len(frames_raw) < 1 or len(frames_raw) > 6:
        raise ValueError("enemy_frames invalid length")
    frames = [bool(v) for v in frames_raw]

    # 2. 契约内的非法取值，按防御式规范化处理（不抛异常）
    enemy_dist_raw = sensor["enemy_dist"]
    if isinstance(enemy_dist_raw, int) and not isinstance(enemy_dist_raw, bool) and enemy_dist_raw >= 0:
        enemy_dist = enemy_dist_raw
    else:
        enemy_dist = 999  # 距离非法，视为远距离

    robot_type_raw = sensor["robot_type"]
    if robot_type_raw == "HERO":
        robot_type = "HERO"
    else:
        robot_type = "INFANTRY"  # 未知机型按步兵处理

    max_hp_raw = sensor["max_hp"]
    if isinstance(max_hp_raw, int) and not isinstance(max_hp_raw, bool) and max_hp_raw > 0:
        max_hp = max_hp_raw
    else:
        max_hp = 1  # 非法 max_hp 设为 1，防止除零

    hp_val = hp if isinstance(hp, int) and not isinstance(hp, bool) else 0
    hp_val = max(0, min(hp_val, max_hp))
    hp_pct = int(hp_val / max_hp * 100)
    hp_pct = max(0, min(100, hp_pct))

    # 术语计算
    visible = frames[-1] if frames else False

    # --- 决策规则 (R1 到 R7，固定顺序，首条命中即返回) ---

    # R1 (保命优先): hp_pct <= 30 时返回 ("RETREAT", RETREAT)
    if hp_pct <= 30:
        return ("RETREAT", SentryState.RETREAT)

    # R2 (撤退保持): state == RETREAT 时
    if state == SentryState.RETREAT:
        if hp_pct > 30:
            return ("RETURN", SentryState.RETURN)
        else:
            return ("RETREAT", SentryState.RETREAT)

    # R3 (返航单帧): state == RETURN 时
    if state == SentryState.RETURN:
        return ("MOVE_BASE", SentryState.PATROL)

    # R4 (交火决策): state == ENGAGE 且可见时
    if state == SentryState.ENGAGE and visible:
        if enemy_dist <= 3:
            return ("SHOOT", SentryState.ENGAGE)
        else:
            if robot_type == "HERO":
                return ("MOVE_RIGHT", SentryState.ENGAGE)
            else:
                return ("MOVE_LEFT", SentryState.ENGAGE)

    # R5 (交火保持): state == ENGAGE 且不可见时
    if state == SentryState.ENGAGE and not visible:
        # 连续两帧不可见视为持续丢失
        if len(frames) >= 2 and not frames[-1] and not frames[-2]:
            return ("SCAN", SentryState.SUSPECT)
        else:
            return ("HOLD_FIRE", SentryState.ENGAGE)

    # R6 (敌情确认): state 为 PATROL 或 SUSPECT 且可见时
    if state in (SentryState.PATROL, SentryState.SUSPECT) and visible:
        # 当前帧与其前一帧均为真，才判定为确认
        if len(frames) >= 2 and frames[-1] and frames[-2]:
            if enemy_dist <= 3:
                return ("SHOOT", SentryState.ENGAGE)
            else:
                if robot_type == "HERO":
                    return ("MOVE_RIGHT", SentryState.ENGAGE)
                else:
                    return ("MOVE_LEFT", SentryState.ENGAGE)
        else:
            # 虽然可见，但未连续两帧确认
            return ("SCAN", SentryState.SUSPECT)

    # R7 (默认行为): state 为 PATROL 或 SUSPECT 且不可见时
    if state in (SentryState.PATROL, SentryState.SUSPECT) and not visible:
        if state == SentryState.PATROL:
            return ("PATROL_MOVE", SentryState.PATROL)
        else:
            return ("SCAN", SentryState.SUSPECT)

    # 理论上不会执行到这里，保底返回
    return ("PATROL_MOVE", SentryState.PATROL)

def _bfs_next_step(start, target, obstacles, width, height):
    """从 start 到 target 的 BFS，返回 start 的下一步方向。"""
    if start == target:
        return None
    q = deque([start])
    parent = {start: None}
    while q:
        cur = q.popleft()
        if cur == target:
            break
        x, y = cur
        for f in (Facing.UP, Facing.RIGHT, Facing.DOWN, Facing.LEFT):
            dx, dy = f.value
            nxt = (x + dx, y + dy)
            if nxt in obstacles or not (0 <= nxt[0] < width and 0 <= nxt[1] < height):
                continue
            if nxt not in parent:
                parent[nxt] = cur
                q.append(nxt)

    if target not in parent:
        return None

    cur = target
    while parent[cur] != start:
        cur = parent[cur]

    dx = cur[0] - start[0]
    dy = cur[1] - start[1]
    for f in (Facing.UP, Facing.RIGHT, Facing.DOWN, Facing.LEFT):
        if f.value == (dx, dy):
            return f
    return None

def run_patrol(grid, max_steps=500):
    """
    Q6: 巡逻任务主循环
    循环 sense -> decide -> act
    返回: {"steps": int, "collisions": int, "visited_count": int, "found_enemy": bool, "success": bool}
    """
    visited = {grid.current_pos}
    steps = 0
    collisions = 0
    consecutive_collisions = 0  

    while steps < max_steps and grid.fuel > 0:
        if grid.current_pos == grid.enemy_pos:
            break

        # 贪心 + 脱困：连续碰撞 3 次后改用 BFS 脱困
        if consecutive_collisions >= 3:
            direction = _bfs_next_step(
                grid.current_pos, grid.enemy_pos,
                grid.obstacles, grid.width, grid.height
            )
            if direction is None:
                direction = next_step_toward(
                    grid.current_pos, grid.enemy_pos,
                    grid.obstacles, grid.facing
                )
        else:
            direction = next_step_toward(
                grid.current_pos, grid.enemy_pos,
                grid.obstacles, grid.facing
            )

        # 对齐朝向（原地转向消耗不计入 steps）
        while grid.facing != direction:
            grid.turn_right()

        old_pos = grid.current_pos
        grid.move_forward()
        steps += 1

        if grid.current_pos == old_pos:
            # 发生碰撞
            collisions += 1
            consecutive_collisions += 1
        else:
            # 成功移动
            consecutive_collisions = 0
            visited.add(grid.current_pos)

    success = (grid.current_pos == grid.enemy_pos)
    
    return {
        "steps": steps,
        "collisions": collisions,
        "visited_count": len(visited),
        "found_enemy": success,
        "success": success
    }

def segment_length_cm(p1, p2):
    """两个检查点 (x, y) 之间的路线长度，单位：厘米。
    检查点坐标单位为格，1 格 = 1 米 = 100 厘米，路线按曼哈顿距离计算。"""
    return (abs(p1[0] - p2[0]) + abs(p1[1] - p2[1])) * 100


def total_route_meters(points):
    """整条巡逻路线的长度，单位：米。
    points 为检查点序列 [(x, y), ...]，至少两个点。"""
    distance_cm = 0
    for i in range(len(points) - 1):
        distance_cm += segment_length_cm(points[i], points[i + 1])
    return distance_cm / 100

def parse_event(line):
    """解析一行事件日志，形如 "MOVE,3" / "SCAN,0" / "IDLE,1"。
    合法返回 {"type": str, "count": int}；脏行返回 None（不得抛异常）。"""
    parts = line.strip().split(",")
    if len(parts) != 2 or parts[0] not in ("MOVE", "SCAN", "IDLE"):
        return None
    if not parts[1].isdigit():
        return None
    return {"type": parts[0], "count": int(parts[1])}


def first_positive(samples):
    """返回样本序列中第一个正数；若没有正数，返回 None（上游约定）。"""
    for s in samples:
        if s > 0:
            return s
    return None


def calibrate(samples):
    """以第一个正样本为基线计算累计漂移：sum(s - baseline)。
    样本为空或没有正样本时，漂移为 0。"""
    baseline = first_positive(samples)
    if baseline is None:
        return 0

    drift = 0
    for s in samples:
        drift += s - baseline
    return drift


def summarize_events(events, max_id):
    """统计 id 不超过 max_id 的事件。

    events: [{"id": int, "move": int, "samples": [int, ...]}, ...]
    每个事件的步数贡献 = move + calibrate(samples)。
    返回 {"events": 统计到的事件数, "steps": 步数总和}。
    """
    used = 0
    steps = 0
    for e in events:
        if e["id"] <= max_id:
            used += 1
            steps += e["move"] + calibrate(e["samples"])
    return {"events": used, "steps": steps}


def log(message, history=None):
    """向历史追加一条日志并返回整个历史列表。
    不显式传入 history 时，每次调用都从空历史开始。"""
    if history is None:
        history = []
    history.append(message)
    return history




def run_legacy_sim(rounds, stamina_start=100):
    """旧版巡逻模拟。

    契约（验收以本条为准）：
    - 依次执行第 0 .. rounds-1 轮，每轮基础消耗 8 点体力；
    - 第 4 轮起（轮号 >= 3）每轮额外消耗 5 点；
    - 任一轮结束后体力 <= 20 时立即终止，不再继续后面的轮；
    - trace 记录 (轮号, 该轮结束时的体力)；
    - 返回 {"rounds": 已执行轮数, "stamina": 剩余体力, "trace": trace}。
    """
    stamina = stamina_start
    round_ = 0
    trace = []

    while round_ < rounds:
        stamina -= 8
        if round_ >= 3:
            stamina -= 5

        trace.append((round_, stamina))

        if stamina <= 20:
            break

        round_ += 1

    return {"rounds": len(trace), "stamina": stamina, "trace": trace}
